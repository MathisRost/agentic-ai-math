\begin{frame}[t]{Why manage context and communication?}
  \noindent\begin{minipage}{\textwidth}
  \raggedright\fontsize{9}{10.4}\selectfont
  An LLM produces text $Y=f_{\theta}(p,c;\xi)$ whose distribution
  depends strongly on the context.\par
  \vspace{1.5mm}
  {\bfseries 1. Keep context focused}\quad
  {\fontsize{7}{8}\selectfont
    \href{https://aclanthology.org/2024.tacl-1.9/}{\color{muted}Liu et al. (2024)};
    \href{https://arxiv.org/html/2502.05167v3}{\color{muted}Modarressi et al. (2025)}}\par
  {\fontsize{8.5}{10}\selectfont%
    \textbf{Problems:} context distraction and context clash.\par}
  Give each agent the information it needs; long histories can obscure relevant details.\par
  \vspace{1mm}

  {\bfseries 2. Give review a fresh context}\quad
  {\fontsize{7}{8}\selectfont
    \href{https://arxiv.org/html/2310.13548v4}{\color{muted}Sharma et al. (2023)};
    \href{https://arxiv.org/html/2309.11495v2}{\color{muted}Dhuliawala et al. (2023)}}\par
  {\fontsize{8.5}{10}\selectfont%
    \textbf{Problems:} context poisoning and context confusion.\par}
  Review the draft and criteria without the writer's history; verify independently.\par
  \vspace{1mm}

  {\bfseries 3. Match the model to the task}\quad
  {\fontsize{7}{8}\selectfont
    \href{https://arxiv.org/abs/2406.18665}{\color{muted}Ong et al., \emph{RouteLLM} (2024)}}\par
  Routing between stronger and cheaper models can improve the balance of cost and quality.\par
  \vspace{1mm}

  {\bfseries 4. Share findings and avoid duplicate work}\quad
  {\fontsize{7}{8}\selectfont
    \href{https://arxiv.org/abs/2305.14325}{\color{muted}Du et al. (2023)};
    \href{https://aclanthology.org/2025.acl-long.1170/}{\color{muted}Wang et al. (2025)}}\par
  Share useful results and redirect work that adds no new information.\par
  \vspace{1mm}

  {\bfseries 5. Explore distinct approaches independently}\quad
  {\fontsize{7}{8}\selectfont
    \href{https://aclanthology.org/2026.findings-acl.1694/}{\color{muted}Zhu et al. (2026)}}\par
  Especially important for mathematics: different starting points can reveal
  new proof strategies.\par
  \end{minipage}
  \vspace{1mm}

  {\fontsize{7}{8.5}\selectfont\color{muted}%
    Effects depend on the task and models. Fresh-context review is a design inference from bias and verification studies.\par}
\end{frame}

\begin{frame}[t]{Point processes: a Poisson example}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.65\textwidth}
      \raggedright\fontsize{9.5}{11.5}\selectfont
      A \textbf{simple point process} $X$ is a random set of distinct
      points in $\mathbb R^d$, finite in bounded regions.\par
      \vspace{1mm}
      Its count in a region $B$ is $X(B)=\#(X\cap B)$.\par
      \vspace{3mm}
      {\bfseries Stationarity}\par
      Shifting all points leaves the distribution unchanged:\par
      \vspace{1mm}
      \noindent\makebox[\linewidth][c]{$\displaystyle X+t\overset{d}{=}X
        \quad\text{for every }t\in\mathbb R^d.$}\par
      \vspace{3mm}
      {\bfseries Homogeneous Poisson process, intensity $\lambda>0$}\par
      \vspace{1mm}
      \noindent\makebox[\linewidth][c]{$\displaystyle
        X(B)\sim\operatorname{Poisson}(\lambda|B|),\qquad
        \mathbb E[X(B)]=\lambda|B|.$}\par
      \vspace{1mm}
      Here $|B|$ is the volume of $B$; $\lambda$ is the mean number
      of points per unit volume.\par
      \vspace{1mm}
      Counts in disjoint regions are independent.\par
    \end{column}
    \begin{column}{0.31\textwidth}
      \centering
      \includegraphics[clip,width=\linewidth,height=48mm,keepaspectratio]{poisson_teaching.pdf}\par
      {\fontsize{7.5}{9}\selectfont\color{muted}
        Unit-area windows; $\lambda=20$.\par
        On average 20 points per window.\par}
    \end{column}
  \end{columns}
  \vspace{1.5mm}
  {\fontsize{7}{8}\selectfont\color{muted}
    \href{https://leman.stat.vt.edu/VTCourses/BaddeleyPointProcesses.pdf}{Baddeley, Sections 1.4--1.6}.
    Pictures show independent simulated realisations in two dimensions.\par}
\end{frame}

\begin{frame}[t]{Product densities: counting pairs}
  \raggedright\fontsize{9.5}{11}\selectfont
  For $n\geq 1$, the \textbf{factorial moment measure} is defined by\par
  \vspace{1mm}
  \noindent\makebox[\textwidth][c]{$\displaystyle
    \alpha^{(n)}(A):=\mathbb E
      \sum\nolimits_{(x_1,\ldots,x_n)\in X^n_{\ne}}
      \mathbf 1_A(x_1,\ldots,x_n),
    \quad A\subseteq(\mathbb R^d)^n\text{ Borel}.$}\par
  \vspace{1.5mm}
  Here $X^n_{\ne}$ consists of all ordered tuples of distinct points of $X$.\par
  \vspace{1mm}
  If $\alpha^{(n)}$ has a density, its \textbf{product density} $\rho^{(n)}$ satisfies\par
  \vspace{0.5mm}
  \noindent\makebox[\textwidth][c]{$\displaystyle
    \alpha^{(n)}(A)=\int_A\rho^{(n)}(x_1,\ldots,x_n)\,\mathrm dx_1\cdots\mathrm dx_n.$}\par
  \vspace{1mm}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \textbf{Example: pairs in two disjoint regions.}\par
      $X(B_1)X(B_2)$ counts pairs with one point in each region.\par
      Its expectation is $\alpha^{(2)}(B_1\times B_2)$.
    \end{column}
    \begin{column}{0.49\textwidth}
      \centering
      \includegraphics[clip,width=\linewidth,height=18mm,keepaspectratio]{product_density_pairs.pdf}
    \end{column}
  \end{columns}
  \vspace{0.5mm}
  For small regions around $x_1,x_2$, where the density varies little:\par
  \vspace{0.5mm}
  \noindent\makebox[\textwidth][c]{$\displaystyle
    \mathbb E[X(B_1)X(B_2)]
    \;\approx\;
    \rho^{(2)}(x_1,x_2)\,|B_1|\,|B_2|.$}\par
  \vspace{1mm}
  {\fontsize{7}{8.5}\selectfont\color{muted}
    \href{https://leman.stat.vt.edu/VTCourses/BaddeleyPointProcesses.pdf}{Baddeley, Section 2.3};

\begin{frame}[t]{Product densities: comparison with Poisson}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.67\textwidth}
      \raggedright\fontsize{9.5}{11}\selectfont
      For a homogeneous Poisson process, independence gives\par
      \vspace{1.5mm}
      \noindent\makebox[\linewidth][c]{$\displaystyle
        \rho^{(2)}(x_1,x_2)=\lambda^2,\qquad
        \rho^{(n)}(x_1,\ldots,x_n)=\lambda^n.$}\par
      \vspace{1.5mm}
      For a stationary process with intensity $\lambda>0$,
      compare its product density with this reference:\par
      \vspace{1.5mm}
      \noindent\makebox[\linewidth][c]{$\displaystyle
        g_n(x_1,\ldots,x_n)=\frac{\rho^{(n)}(x_1,\ldots,x_n)}{\lambda^n}.$}\par
      \vspace{1mm}
      This is the \textbf{$n$-point correlation function}.\par
      \vspace{1mm}
      Compared with Poisson at the same intensity:\par
      \vspace{1mm}
      {$g_n>1$: more such tuples.}\par
      {$g_n<1$: fewer such tuples.}\par
      $g_n=1$: the same local tuple density.\par
      \vspace{1mm}
      $n=2$: pairs; $n=3$: triangles; $n=4$: tetrahedra.\par
    \end{column}
    \begin{column}{0.30\textwidth}
      \centering
      \includegraphics[clip,width=\linewidth,height=54mm,keepaspectratio]{point_process_patterns.pdf}\par
    \end{column}
  \end{columns}
  \vspace{0.5mm}
  {\fontsize{7}{8.5}\selectfont\color{muted}
    Poisson simulated; other patterns schematic. Comparisons are local; $g_2=1$ alone does not imply Poisson.\par
    \href{https://leman.stat.vt.edu/VTCourses/BaddeleyPointProcesses.pdf}{Baddeley, Section 2.3};
    \href{https://stoch.math.kit.edu/img/Last/lastpenrose2017.pdf}{Last and Penrose, Section 4.4}.\par}
\end{frame}

\begin{frame}[t]{Step 1: the precise problem}
  \noindent\begin{minipage}{\textwidth}
  \raggedright\fontsize{9.5}{11}\selectfont
  Fix $M,\lambda_0,h_3,h_4>0$.\par
  Find one stationary, simple point process $X$ on $\mathbb R^3$ with intensity
  $\lambda_0$, finite local moments through order four and product densities.\par
  \vspace{1mm}
  \noindent\makebox[\textwidth][c]{%
    \includegraphics[clip,width=0.31\textwidth,height=19mm,keepaspectratio]{simplex_segment.pdf}\hfill
    \includegraphics[clip,width=0.31\textwidth,height=19mm,keepaspectratio]{simplex_triangle.pdf}\hfill
    \includegraphics[clip,width=0.31\textwidth,height=19mm,keepaspectratio]{simplex_tetrahedron.pdf}}\par
  \vspace{1mm}
  For some constants $C_2,C_3,C_4>0$, require simultaneously:\par
  \vspace{1mm}
  \noindent\makebox[\linewidth][c]{$\displaystyle
    g_2(x_1,x_2)=\frac{C_2}{D},
    \qquad
    g_3(x_1,x_2,x_3)=\frac{C_3}{A},
    \qquad
    g_4(x_1,x_2,x_3,x_4)=\frac{C_4}{V}.
  $}\par
  \vspace{1mm}
  Almost everywhere on nondegenerate configurations of diameter $\le M$,
  with every triangle altitude $\ge h_3$ and every tetrahedron altitude $\ge h_4$.\par
  \end{minipage}
  \vspace{1mm}
  {\fontsize{7}{8.5}\selectfont\color{muted}
    The conjecture is motivated by observed spatial correlations in earthquake catalogues.\par
    \href{https://doi.org/10.1111/j.1365-246X.1980.tb04857.x}{Kagan and Knopoff (1980)};
    Kagan (1981):
    \href{https://doi.org/10.1111/j.1365-246X.1981.tb06948.x}{three-point}
    and \href{https://doi.org/10.1111/j.1365-246X.1981.tb06949.x}{four-point studies}.\par}
\end{frame}

\begin{frame}[t]{Results and personal feedback}
  \vspace{-3mm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.45\textwidth}
      {\fontsize{11}{13}\selectfont\bfseries\color{teal}Poisson cluster model\par}
      \vspace{2mm}
      {\raggedright\fontsize{9}{10.5}\selectfont
      Homogeneous Poisson parents $\Pi$ each generate 0--3 children,
      with jointly chosen offsets $C_z$:\par
      \vspace{0.5mm}
      \noindent\makebox[\linewidth][c]{$\textstyle
        X=\bigcup\nolimits_{z\in\Pi}\bigl(\{z\}\cup(z+C_z)\bigr).$}\par
      \vspace{0.5mm}
      $C_z$ contains only child offsets. The parent is retained,
      giving 1--4 points per cluster.\par
      \vspace{0.5mm}
      \textbf{How many points?} The solution gives a probability distribution
      over cluster sizes $j=1,\ldots,4$.\par
      \vspace{0.5mm}
      \textbf{Where are they placed?} For $j\geq2$, the normalised density\par
      \vspace{0.5mm}
      \noindent\makebox[\linewidth][c]{$\displaystyle
        \frac{F_j(z,x_1,\ldots,x_{j-1})}{Z_j}$}\par
      \vspace{0.5mm}
      describes how the $j-1$ children are jointly distributed around
      the parent $z$.\par}
    \end{column}

\begin{frame}[t]{Future of agentic research}
  \vspace{-1mm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.45\textwidth}
      {\fontsize{10.5}{12}\selectfont\bfseries
        What I expect to change\par}
      \vspace{2mm}
      {\raggedright\fontsize{9}{10.8}\selectfont
        Stronger models can increasingly plan, divide work into tasks,
        use tools and delegate to subagents.\par
        \vspace{0.8mm}
        For a precise mathematical problem, one initial prompt may
        increasingly suffice. This can still mean substantial autonomous
        work, with no guarantee of success.\par
        \vspace{0.8mm}
        As platforms provide more harness functions, we will have
        less to build ourselves.\par
        \vspace{0.8mm}
        I also expect explanations to become less sloppy and more
        useful for learning.\par}
    \end{column}
    \begin{column}{0.51\textwidth}
      {\fontsize{10.5}{12}\selectfont\bfseries
        Where a custom harness still helps\par}
      \vspace{2mm}
      {\fontsize{8.8}{10.5}\selectfont
      \begin{itemize}
        \setlength{\itemsep}{0.5mm}
        \item Keep the exact target, hypotheses and criteria for success fixed.
        \item Require separate review and formal verification.
        \item Keep human checkpoints for direction and interpretation.
        \item Record results, failed approaches, sources and dependencies.
        \item Set compute budgets, stopping rules and model choice.
        \item When writing papers with AI, set your preferences for writing
          and illustration styles.
      \end{itemize}}
    \end{column}
  \end{columns}
